# Object detection for demos/scavenger_hunt. Ultralytics ships COCO-pretrained
# weights and downloads them on first use, so no model file has to be supplied.
#
# torch arrives as an ultralytics dependency. Installing plainly from PyPI pulls
# the CUDA build and roughly 3-5 GB of nvidia wheels; the CPU build is ~250 MB and
# is all these demos need (see docs/PRD.md: CPU-friendly inference). Install it
# first from PyTorch's CPU index, then this file:
#
#   pip install --index-url https://download.pytorch.org/whl/cpu torch torchvision
#   pip install -r requirements.txt
#
# Omit that first line if you want GPU acceleration and have a CUDA setup.
ultralytics==8.4.111
torch==2.13.0
torchvision==0.28.0
