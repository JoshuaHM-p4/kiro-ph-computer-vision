{% endblock %} {% block script %} import { LandmarkStream, bindStatusUi, fitCanvas } from "{{ url_for('shared.static', filename='js/landmark-stream.js') }}"; const video = document.getElementById('cam'); const canvas = document.getElementById('overlay'); const ctx = canvas.getContext('2d'); fitCanvas(canvas); const setText = (id, value) => { document.getElementById(id).textContent = value; }; function draw(state) { ctx.clearRect(0, 0, canvas.width, canvas.height); const trail = state.trail || []; if (trail.length > 1) { ctx.strokeStyle = '#4ce7ff'; ctx.lineWidth = 4; ctx.lineJoin = 'round'; ctx.beginPath(); trail.forEach((p, i) => { const x = p.x * canvas.width, y = p.y * canvas.height; if (i === 0) ctx.moveTo(x, y); else ctx.lineTo(x, y); }); ctx.stroke(); } if (state.point) { ctx.fillStyle = '#8cff8c'; ctx.beginPath(); ctx.arc(state.point.x * canvas.width, state.point.y * canvas.height, 10, 0, Math.PI * 2); ctx.fill(); } setText('r-frames', state.frames ?? 0); setText('r-hands', state.hands ?? 0); setText('r-accepted', state._meta?.accepted ?? 0); setText('r-dropped', state._meta?.dropped ?? 0); } const stream = new LandmarkStream({ video, basePath: '/echo', need: { hands: true }, numHands: 1, onState: draw, onStatus: bindStatusUi('conn-pill', 'status'), }); document.getElementById('btn-clear') .addEventListener('click', () => stream.sendCommand('clear')); document.getElementById('btn-snap') .addEventListener('click', (e) => { e.target.href = stream.snapshotUrl(); }); stream.start().catch((err) => { document.getElementById('status').innerHTML = `${err.message} — camera permission and internet access are required.`; }); {% endblock %}
